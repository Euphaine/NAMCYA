@code {
    private string enteredPin = "";
    private string loginError = "";
    private string successMessage = "";
    private bool isLoggedIn = false;

    [SupplyParameterFromQuery]
    public int? EventId { get; set; }

    [SupplyParameterFromQuery]
    public string? Pin { get; set; }

    private Judge? currentJudge;
    private EventItem? currentEvent;
    private List<Contestant> contestants = new();
    private List<Criterion> criteria = new();
    private List<Score> existingScores = new();

    protected override async Task OnParametersSetAsync()
    {
        // Automatically trigger if query parameters are present and we aren't already logged in
        if (!isLoggedIn && EventId.HasValue && !string.IsNullOrEmpty(Pin))
        {
            await AttemptAutoLogin(EventId.Value, Pin);
        }
    }

    private async Task AttemptAutoLogin(int eventId, string pin)
    {
        loginError = "";
        
        currentJudge = await Db.Judges
            .FirstOrDefaultAsync(j => j.EventId == eventId && j.PinCode == pin);

        if (currentJudge == null)
        {
            loginError = "Invalid QR login credentials.";
            return;
        }

        currentEvent = await Db.Events.FindAsync(currentJudge.EventId);
        if (currentEvent == null)
        {
            loginError = "Associated event not found.";
            return;
        }

        // Load the necessary scoring datasets
        contestants = await Db.Contestants.Where(c => c.EventId == currentEvent.Id).OrderBy(c => c.ContestantNumber).ToListAsync();
        criteria = await Db.Criteria.Where(cr => cr.EventId == currentEvent.Id).ToListAsync();
        existingScores = await Db.Scores.Where(s => s.JudgeId == currentJudge.Id).ToListAsync();

        isLoggedIn = true;
    }

    private async Task AuthenticateJudge()
    {
        loginError = "";
        successMessage = "";
        if (string.IsNullOrWhiteSpace(enteredPin))
        {
            loginError = "Please enter your PIN code.";
            return;
        }

        currentJudge = await Db.Judges.FirstOrDefaultAsync(j => j.PinCode == enteredPin);
        if (currentJudge == null)
        {
            loginError = "Invalid PIN code. Please check with your event administrator.";
            return;
        }

        currentEvent = await Db.Events.FindAsync(currentJudge.EventId);
        if (currentEvent == null)
        {
            loginError = "Associated event not found.";
            return;
        }

        contestants = await Db.Contestants.Where(c => c.EventId == currentEvent.Id).OrderBy(c => c.ContestantNumber).ToListAsync();
        criteria = await Db.Criteria.Where(cr => cr.EventId == currentEvent.Id).ToListAsync();
        existingScores = await Db.Scores.Where(s => s.JudgeId == currentJudge.Id).ToListAsync();

        isLoggedIn = true;
    }

    private void Logout()
    {
        isLoggedIn = false;
        enteredPin = "";
        successMessage = "";
        currentJudge = null;
        currentEvent = null;
        contestants.Clear();
        criteria.Clear();
        existingScores.Clear();
        
        // Clear query parameters from URL cleanly upon logout
        Navigation.NavigateTo("/judge-portal", forceLoad: false);
    }

    private string GetFormattedScore(int contestantId, int criterionId)
    {
        var scoreRecord = existingScores.FirstOrDefault(s => s.ContestantId == contestantId && s.CriterionId == criterionId);
        if (scoreRecord == null) return "0";
        return scoreRecord.ScoreValue.ToString(CultureInfo.InvariantCulture);
    }

    private async Task UpdateScore(int contestantId, int criterionId, object? val)
    {
        if (currentJudge == null || currentEvent == null) return;

        string stringVal = val?.ToString() ?? "0";
        stringVal = stringVal.Replace(',', '.');

        if (decimal.TryParse(stringVal, NumberStyles.Any, CultureInfo.InvariantCulture, out decimal parsedScore))
        {
            var scoreRecord = existingScores.FirstOrDefault(s => s.ContestantId == contestantId && s.CriterionId == criterionId);

            if (scoreRecord != null)
            {
                scoreRecord.ScoreValue = parsedScore;
                Db.Scores.Update(scoreRecord);
            }
            else
            {
                var newScore = new Score
                {
                    EventId = currentEvent.Id,
                    JudgeId = currentJudge.Id,
                    ContestantId = contestantId,
                    CriterionId = criterionId,
                    ScoreValue = parsedScore
                };
                Db.Scores.Add(newScore);
                existingScores.Add(newScore);
            }

            await Db.SaveChangesAsync();
        }
    }

    private string GetLocalIpAddress()
{
    try
    {
        foreach (var networkInterface in NetworkInterface.GetAllNetworkInterfaces())
        {
            if (networkInterface.OperationalStatus == OperationalStatus.Up &&
                networkInterface.NetworkInterfaceType != NetworkInterfaceType.Loopback &&
                networkInterface.NetworkInterfaceType != NetworkInterfaceType.Tunnel)
            {
                string name = networkInterface.Name.ToLower();
                string desc = networkInterface.Description.ToLower();

                if (name.Contains("vethernet") || desc.Contains("virtual") || desc.Contains("hyper-v") || desc.Contains("subsystem"))
                {
                    continue;
                }

                var properties = networkInterface.GetIPProperties();
                foreach (var addressInfo in properties.UnicastAddresses)
                {
                    if (addressInfo.Address.AddressFamily == AddressFamily.InterNetwork)
                    {
                        string ipStr = addressInfo.Address.ToString();
                        if (ipStr.StartsWith("192.168.") || ipStr.StartsWith("10.") || ipStr.StartsWith("172."))
                        {
                            return ipStr;
                        }
                    }
                }
            }
        }
    }
    catch
    {
        // Fallback safety
    }

    return "127.0.0.1";
}

private void GoToPersonalRanking()
{
    if (currentEvent != null && currentJudge != null)
    {
        string localIp = GetLocalIpAddress();
        // Pointing directly to the judge portal route with session/auth parameters ensures seamless auto-login context
        string url = $"http://{localIp}:5000/judge-portal?eventId={currentEvent.Id}&pin={currentJudge.PinCode}&redirect=ranking";
        Navigation.NavigateTo(url, forceLoad: true);
    }
}

    private void SubmitScores()
    {
        successMessage = "Scores successfully finalized and submitted!";
    }
