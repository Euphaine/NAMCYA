    protected override void OnInitialized()
    {
        Navigation.ToAbsoluteUri("/images/banner.png");
        Navigation.ToAbsoluteUri("/images/logo.png");
    }

    protected override async Task OnParametersSetAsync()
    {
        if (!isLoggedIn && EventId.HasValue && !string.IsNullOrEmpty(Pin))
        {
            await AttemptAutoLogin(EventId.Value, Pin);
        }
    }

    private async Task AttemptAutoLogin(int eventId, string pin)
    {
        loginError = "";
        
        currentJudge = await Db.Judges
            .FirstOrDefaultAsync(j => j.EventId == eventId && j.PinCode == pin);

        if (currentJudge == null)
        {
            loginError = "Invalid QR login credentials.";
            return;
        }

        currentEvent = await Db.Events.FindAsync(currentJudge.EventId);
        if (currentEvent == null)
        {
            loginError = "Associated event not found.";
            return;
        }

        contestants = await Db.Contestants.Where(c => c.EventId == currentEvent.Id).OrderBy(c => c.ContestantNumber).ToListAsync();
        criteria = await Db.Criteria.Where(cr => cr.EventId == currentEvent.Id).ToListAsync();
        existingScores = await Db.Scores.Where(s => s.JudgeId == currentJudge.Id).ToListAsync();

        isLoggedIn = true;
        showWelcomeOverlay = true;
    }

    private async Task AuthenticateJudge()
    {
        loginError = "";
        successMessage = "";
        if (string.IsNullOrWhiteSpace(enteredPin))
        {
            loginError = "Please enter your PIN code.";
            return;
        }

        currentJudge = await Db.Judges.FirstOrDefaultAsync(j => j.PinCode == enteredPin);
        if (currentJudge == null)
        {
            loginError = "Invalid PIN code. Please check with your event administrator.";
            return;
        }

        currentEvent = await Db.Events.FindAsync(currentJudge.EventId);
        if (currentEvent == null)
        {
            loginError = "Associated event not found.";
            return;
        }

        contestants = await Db.Contestants.Where(c => c.EventId == currentEvent.Id).OrderBy(c => c.ContestantNumber).ToListAsync();
        criteria = await Db.Criteria.Where(cr => cr.EventId == currentEvent.Id).ToListAsync();
        existingScores = await Db.Scores.Where(s => s.JudgeId == currentJudge.Id).ToListAsync();

        isLoggedIn = true;
        showWelcomeOverlay = true;
    }

    private async Task DismissWelcomeOverlay()
    {
        isOverlayFadingOut = true;
        StateHasChanged();
        await Task.Delay(400);
        showWelcomeOverlay = false;
        isOverlayFadingOut = false;
        StateHasChanged();
    }

    private void TogglePersonalRanking()
    {
        showPersonalRanking = !showPersonalRanking;
    }

    private void Logout()
    {
        isLoggedIn = false;
        showWelcomeOverlay = false;
        showPersonalRanking = false;
        enteredPin = "";
        successMessage = "";
        currentJudge = null;
        currentEvent = null;
        contestants.Clear();
        criteria.Clear();
        existingScores.Clear();
        
        Navigation.NavigateTo("/judge-portal", forceLoad: false);
    }

    private string GetFormattedScore(int contestantId, int criterionId)
    {
        var scoreRecord = existingScores.FirstOrDefault(s => s.ContestantId == contestantId && s.CriterionId == criterionId);
        if (scoreRecord == null) return "0";
        return scoreRecord.ScoreValue.ToString(CultureInfo.InvariantCulture);
    }

    private void UpdateScore(int contestantId, int criterionId, object? val)
    {
        if (currentJudge == null || currentEvent == null) return;

        string stringVal = val?.ToString() ?? "0";
        stringVal = stringVal.Replace(',', '.');

        if (decimal.TryParse(stringVal, NumberStyles.Any, CultureInfo.InvariantCulture, out decimal parsedScore))
        {
            var scoreRecord = existingScores.FirstOrDefault(s => s.ContestantId == contestantId && s.CriterionId == criterionId);

            if (scoreRecord != null)
            {
                scoreRecord.ScoreValue = parsedScore;
            }
            else
            {
                var newScore = new Score
                {
                    EventId = currentEvent.Id,
                    JudgeId = currentJudge.Id,
                    ContestantId = contestantId,
                    CriterionId = criterionId,
                    ScoreValue = parsedScore
                };
                existingScores.Add(newScore);
            }
        }
    }

    private async Task SubmitScores()
    {
        if (currentJudge == null || currentEvent == null) return;

        try
        {
            foreach (var scoreRecord in existingScores)
            {
                var dbRecord = await Db.Scores.FirstOrDefaultAsync(s => 
                    s.JudgeId == currentJudge.Id && 
                    s.ContestantId == scoreRecord.ContestantId && 
                    s.CriterionId == scoreRecord.CriterionId);

                if (dbRecord != null)
                {
                    dbRecord.ScoreValue = scoreRecord.ScoreValue;
                    Db.Scores.Update(dbRecord);
                }
                else
                {
                    Db.Scores.Add(scoreRecord);
                }
            }

            await Db.SaveChangesAsync();
            successMessage = "Scores successfully finalized and submitted to the server!";
        }
        catch (Exception ex)
        {
            successMessage = $"Error saving scores: {ex.Message}";
        }
    }
}
