@code {
    private int totalEvents = 0;
    private int totalContestants = 0;
    private int totalJudges = 0;
    private List<EventItem> events = new();
    private List<Judge> judges = new();
    private string localIp = "127.0.0.1";
    private string qrCodeSvg = "";

    protected override async Task OnInitializedAsync()
    {
        localIp = GetLocalIpAddress();

        string portalUrl = $"http://{localIp}:5000/judge-portal";
        var qr = QrCode.EncodeText(portalUrl, QrCode.Ecc.Medium);
        qrCodeSvg = qr.ToSvgString(2);

        totalEvents = await Db.Events.CountAsync();
        totalContestants = await Db.Contestants.CountAsync();
        totalJudges = await Db.Judges.CountAsync();
        
        events = await Db.Events.OrderByDescending(e => e.Id).Take(5).ToListAsync();
        judges = await Db.Judges.ToListAsync();
    }

    private string GetLocalIpAddress()
    {
        try
        {
            foreach (var networkInterface in NetworkInterface.GetAllNetworkInterfaces())
            {
                if (networkInterface.OperationalStatus == OperationalStatus.Up &&
                    networkInterface.NetworkInterfaceType != NetworkInterfaceType.Loopback &&
                    networkInterface.NetworkInterfaceType != NetworkInterfaceType.Tunnel)
                {
                    string name = networkInterface.Name.ToLower();
                    string desc = networkInterface.Description.ToLower();

                    if (name.Contains("vethernet") || desc.Contains("virtual") || desc.Contains("hyper-v") || desc.Contains("subsystem"))
                    {
                        continue;
                    }

                    var properties = networkInterface.GetIPProperties();
                    foreach (var addressInfo in properties.UnicastAddresses)
                    {
                        if (addressInfo.Address.AddressFamily == AddressFamily.InterNetwork)
                        {
                            string ipStr = addressInfo.Address.ToString();
                            if (ipStr.StartsWith("192.168.") || ipStr.StartsWith("10.") || ipStr.StartsWith("172."))
                            {
                                return ipStr;
                            }
                        }
                    }
                }
            }
        }
        catch
        {
            // Fallback safety
        }

        return "127.0.0.1";
    }
}
